-- ========================================================= -- SKLAD42 — подготовка безопасной авторизации -- ========================================================= -- 1. Связываем сотрудника с пользователем Supabase Auth ALTER TABLE public.employees ADD COLUMN IF NOT EXISTS auth_user_id uuid UNIQUE; -- 2. Индекс для быстрого поиска сотрудника по Auth ID CREATE INDEX IF NOT EXISTS idx_employees_auth_user_id ON public.employees(auth_user_id); -- 3. Функция: текущий сотрудник CREATE OR REPLACE FUNCTION public.current_employee() RETURNS public.employees LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$ SELECT e.* FROM public.employees e WHERE e.auth_user_id = auth.uid() AND e.active = true LIMIT 1; $$; -- 4. Функция: ID текущего сотрудника CREATE OR REPLACE FUNCTION public.current_employee_id() RETURNS text LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$ SELECT e.id FROM public.employees e WHERE e.auth_user_id = auth.uid() AND e.active = true LIMIT 1; $$; -- 5. Функция: склад текущего сотрудника -- NULL означает полный доступ ко всем складам CREATE OR REPLACE FUNCTION public.current_warehouse_id() RETURNS text LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$ SELECT e.warehouse_id FROM public.employees e WHERE e.auth_user_id = auth.uid() AND e.active = true LIMIT 1; $$; -- 6. Проверка права текущего сотрудника CREATE OR REPLACE FUNCTION public.has_permission(permission_name text) RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public AS $$ SELECT COALESCE( (e.role_id IS NOT NULL AND COALESCE((r.permissions ->> permission_name)::boolean, false)), false ) FROM public.employees e LEFT JOIN public.roles r ON r.id = e.role_id WHERE e.auth_user_id = auth.uid() AND e.active = true LIMIT 1; $$;
